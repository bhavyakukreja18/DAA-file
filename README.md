# Program 14: Binary Search
## Objective
Write a python program for binary search

## Algorithm
```
Function Main
    Declare Integer i, n, num, low, high, mid

    Input n
    Declare Integer Array numbers[n]

    For i = 0 to n-1
        Input numbers[i]
    End
    Assign low = 0
    Assign high = n-1
    Input num
    While low<high
        Assign mid = (low+high)/2
        If numbers[mid]==num
        Else
            If numbers[mid]<num
                Assign low = mid+1
            Else
                Assign high = mid-1
            End
        End
    End
    Output mid+1
End


```
## Program
```python
def binary_search(arr, target):
    low = 0
    high = len(arr) - 1

    while low <=high:
        mid = (low + high) // 2
        
        if arr[mid] == target:
            return mid  # Target found
        elif arr[mid] < target:
            low = mid + 1  # Search right half
        else:
            high = mid - 1  # Search left half
            
    return -1  


numbers = []
total_elements = int(input("How many numbers do you want to enter? "))

for i in range(total_elements):
    num = int(input("Enter number"))
    numbers.append(num)


numbers.sort()
print("Sorted list for Binary Search", numbers)

target_number = int(input("Enter the number to be searched: "))
result = binary_search(numbers, target_number)

print("Index position", result)



```

```python
def insertion_sort(arr):
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
```

```python
        visited = set()
        queue = deque([start])
        visited.add(start)

        while queue:
            node = queue.popleft()
            print(node.data, end=" ")

            for neighbor in node.neighbors:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
```

# Program 29: Dijkstra Algorithm

## Objective
Write a python program to show Dijkstra Algorithm

## Program
```python
class Node:
    def __init__(self, data):
        self.data = data
        self.neighbors = []


class Graph:
    def __init__(self):
        self.nodes = []

    def add_node(self, data):
        node = Node(data)
        self.nodes.append(node)
        return node

    def add_edge(self, node1, node2, weight):
        node1.neighbors.append((node2, weight))
        node2.neighbors.append((node1, weight))

    def print_graph(self):
        for node in self.nodes:
            print(node.data, "->", end=" ")

            for neighbor, weight in node.neighbors:
                print(f"{neighbor.data}({weight})", end=" ")

            print()

    def dijkstra(self, source):
        distance = {}

        # Step 1: Initialize distances
        for node in self.nodes:
            distance[node] = float('inf')

        distance[source] = 0

        visited = set()

        # Step 2: Repeat until all nodes are visited
        while len(visited) < len(self.nodes):

            # Find minimum-distance unvisited node
            current = None
            min_distance = float('inf')

            for node in self.nodes:
                if node not in visited and distance[node] < min_distance:
                    min_distance = distance[node]
                    current = node

            # No reachable node left
            if current is None:
                break

            # Mark current node visited
            visited.add(current)

            # Relax edges
            for neighbor, weight in current.neighbors:
                new_distance = distance[current] + weight

                if new_distance < distance[neighbor]:
                    distance[neighbor] = new_distance

        return distance
```

```python
        # Step 1: Initialize distances
        distance = {}

        for node in self.nodes:
            distance[node] = float('inf')

        distance[source] = 0

        # Step 2: Relax all edges V-1 times
```
